import { chromium,firefox,webkit } from '/home/abhispra/mvp/website_latest/tribera/node_modules/playwright/index.mjs';
import fs from 'node:fs';
const out='artifacts/v9-sectors/claude-detail', records=[];
for (const [engine,type] of Object.entries({chromium,firefox,webkit})) {
 const browser=await type.launch({headless:true}); const page=await browser.newPage({reducedMotion:'reduce'});
 for(const slug of ['industrial-engineering','manufacturing','pharmaceutical','private-equity-portfolio']) for(const width of engine==='chromium'?[320,390,768,1366,1920]:[390]) {
 await page.setViewportSize({width,height:900});await page.goto('http://localhost:4390/__v9/industries/'+slug);await page.evaluate(()=>document.fonts.ready);
 const state=await page.locator('.faq-item').evaluateAll(items=>items.map(d=>({open:d.open,answerHeight:d.querySelector('.faq-a').getBoundingClientRect().height,answerDisplay:getComputedStyle(d.querySelector('.faq-a')).display})));
 records.push({engine,slug,width,state});
 if(slug==='private-equity-portfolio' && engine==='chromium') await page.locator('.faq-list').screenshot({path:`${out}/faq-${width}.png`});
 }
 await browser.close();
}
fs.writeFileSync(`${out}/faq-state.json`,JSON.stringify(records,null,2));console.log(records.length+' FAQ viewport states inspected');
